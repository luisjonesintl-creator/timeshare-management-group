const SUPABASE_URL = "https://ztojbyfbyidzzrqicjvn.supabase.co"; const SUPABASE_ANON_KEY = "sb_publishable_cYSA9_lak5EnHx-b9Q4SQg_6Z-o0h7f"; const supabaseClientInstance = window.supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY); // Buscador automático asíncrono en la API pública de Wikipedia async function getWikipediaImage(resortName) { if (!resortName) return null; try { const cleanQuery = encodeURIComponent(resortName.trim()); const response = await fetch(`https://wikipedia.org{cleanQuery}&prop=pageimages&format=json&pithumbsize=800&origin=*`); const data = await response.json(); const pages = data.query.pages; const pageId = Object.keys(pages); if (pageId && pages[pageId].thumbnail) { return pages[pageId].thumbnail.source; } } catch (err) { console.warn("Wikipedia Image Lookup skipped for:", resortName); } return null; } async function fetchAndRenderSales() { const container = document.getElementById("active-properties"); if (!container) return; try { // Solicitamos a Supabase únicamente las propiedades registradas para la Venta const { data: properties, error } = await supabaseClientInstance .from('properties') .select('*') .eq('listing_type', 'SALE'); if (error) throw error; if (!properties || properties.length === 0) { container.innerHTML = `
No for-sale intervals available at the moment.
`; return; } container.innerHTML = ""; // Recorremos secuencialmente los resorts resolviendo las fotos en caliente desde la API for (const item of properties) { let wikiImg = await getWikipediaImage(item.resort_name); let finalImageUrl = ""; if (wikiImg) { finalImageUrl = wikiImg; // Wikipedia Oficial } else if (item.image_url && String(item.image_url).startsWith('http')) { finalImageUrl = item.image_url; // Respaldo manual de Supabase } else { finalImageUrl = "https://unsplash.com"; // Respaldo de seguridad } const cardHtml = `
For SALE Week ${item.week_number || 'N/A'}
ID #TMG${item.id}

${item.resort_name || 'Premium Resort'}

Asking Price $${Number(item.asking_price || 0).toLocaleString()}
`; container.innerHTML += cardHtml; } } catch (err) { console.error("Error listings:", err); } } document.addEventListener("DOMContentLoaded", () => { fetchAndRenderSales(); });
