// Función asíncrona máster para consultar, contar y dibujar todo el ecosistema async function runAdminSynchronizer() { const leadsContainer = document.getElementById("admin-leads-list"); const inventoryContainer = document.getElementById("admin-inventory-table"); const statProperties = document.getElementById("stat-properties"); const statLeads = document.getElementById("stat-leads"); const badgeCount = document.getElementById("leads-count-badge"); try { // 1. Extraemos de forma masiva los datos de la tabla 'properties' const { data: properties, error: errProp } = await supabaseClientInstance .from('properties') .select('*'); if (errProp) throw errProp; if (statProperties) statProperties.innerText = properties ? properties.length : "0"; // 2. Extraemos de forma masiva los datos de la tabla 'leads' const { data: leads, error: errLeads } = await supabaseClientInstance .from('leads') .select('*') .order('created_at', { ascending: false }); if (errLeads) throw errLeads; if (statLeads) statLeads.innerText = leads ? leads.length : "0"; if (badgeCount) badgeCount.innerText = `${leads ? leads.length : 0} TOTAL`; // 3. Pintamos la lista de leads en la barra lateral izquierda if (leadsContainer) { if (!leads || leads.length === 0) { leadsContainer.innerHTML = `
No customer inquiries recorded yet.
`; } else { leadsContainer.innerHTML = ""; leads.forEach(lead => { const dateObj = new Date(lead.created_at); const cleanDate = dateObj.toLocaleDateString() + ' ' + dateObj.toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'}); leadsContainer.innerHTML += `
${lead.name} ${cleanDate}
${lead.email}

${lead.message}

`; }); } } // 4. Pintamos la tabla de gestión de inventario en caliente (Nueva sección) if (inventoryContainer) { if (!properties || properties.length === 0) { inventoryContainer.innerHTML = `No active listings inside database.`; return; } inventoryContainer.innerHTML = ""; properties.forEach(item => { const isSale = item.listing_type === 'SALE'; const badgeColor = isSale ? 'bg-blue-950/50 text-blue-400 border-blue-900/40' : 'bg-cyan-950/50 text-cyan-400 border-cyan-900/40'; inventoryContainer.innerHTML += ` ${item.resort_name || 'Premium Asset'} Asset ID: #TMG${item.id} ${item.listing_type || 'SALE'} ${item.week_number || 'N/A'}
$
`; }); } } catch (error) { console.error("Error sincronizando consola administrativa:", error); } } // ====== 5. LÓGICA DE ACTUALIZACIÓN EN CALIENTE DIRECTA A SUPABASE ====== window.updatePriceInCaliente = async function(propertyId) { const priceInput = document.getElementById(`input-price-${propertyId}`); if (!priceInput) return; const newPrice = parseFloat(priceInput.value); if (isNaN(newPrice) || newPrice < 0) { alert("⚠️ Please enter a valid positive pricing amount."); return; } try { // Modificamos el registro de forma instantánea en tu tabla de Supabase const { error } = await supabaseClientInstance .from('properties') .update({ asking_price: newPrice }) .eq('id', propertyId); if (error) throw error; alert("🎯 Price updated in real-time! Public catalog feed successfully synchronized."); runAdminSynchronizer(); // Recargamos los contadores superiores de la interfaz } catch (err) { console.error("Error actualizando precio:", err); alert("⚠️ Transmission stalled. Could not overwrite pricing matrix structure."); } }; // ====== 6. LÓGICA DE BORRADO SEGURO DIRECTO A SUPABASE ====== window.deleteListingInCaliente = async function(propertyId) { const secureConfirm = confirm("⚠️ WARNING: Are you sure you want to permanently delete this resort asset from the live catalog?"); if (!secureConfirm) return; try { // Eliminamos la fila correspondiente usando el ID único del resort const { error } = await supabaseClientInstance .from('properties') .delete() .eq('id', propertyId); if (error) throw error; alert("🗑️ Listing destroyed successfully. Removed from public servers."); // Eliminamos visualmente la fila de la tabla sin necesidad de recargar la pestaña const row = document.getElementById(`property-row-${propertyId}`); if (row) row.remove(); runAdminSynchronizer(); // Sincronizamos las métricas y totales } catch (err) { console.error("Error eliminando propiedad:", err); alert("⚠️ Deletion failure. Check your remote user permission policies."); } }; // Interceptamos el formulario para inyectar nuevos resorts de manera directa document.getElementById("admin-add-property-form").addEventListener("submit", async function(e) { e.preventDefault(); const submitBtn = this.querySelector("button[type='submit']"); const resortName = document.getElementById("new-resort-name").value.trim(); const listingType = document.getElementById("new-listing-type").value; const weekNumber = parseInt(document.getElementById("new-week-number").value); const askingPrice = parseFloat(document.getElementById("new-asking-price").value); const imageUrl = document.getElementById("new-image-url").value.trim(); const recordToInject = { resort_name: resortName, listing_type: listingType, week_number: weekNumber, asking_price: askingPrice, image_url: imageUrl, status: 'AVAILABLE' }; if (submitBtn) { submitBtn.disabled = true; submitBtn.innerText = "Transmitting Database Cluster..."; } try { const { error } = await supabaseClientInstance .from('properties') .insert([recordToInject]); if (error) throw error; alert("🚀 Resort asset successfully injected into properties database table!"); this.reset(); runAdminSynchronizer(); } catch (err) { console.error("Error inyectando registro:", err); alert("⚠️ Insertion stalled. Ensure your columns match exact table structure types."); } finally { if (submitBtn) { submitBtn.disabled = false; submitBtn.innerText = "Inject Listing to Database"; } } }); // Lanzamos la sincronización de la base de datos al cargar el DOM document.addEventListener("DOMContentLoaded", () => { runAdminSynchronizer(); });
